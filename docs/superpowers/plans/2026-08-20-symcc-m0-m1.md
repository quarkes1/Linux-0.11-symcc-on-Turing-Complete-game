def test_add_reg():
    # add r1, r2, r3 → 00100100 00010010 00000011 00000000
    assert t.encode("add r1, r2, r3") == bytes.fromhex("24 12 03 00")

def test_add_imm():
    # add r1, r2, 0x1234 → 00110100 00010010 00110100 00010000 → wait, 字段位序见 .isa
    # add %a(reg), %b(reg), %c:U16(imm): a=bits16-19, b=bits12-15, c=bits0-15
    # r1=0001, r2=0010 → aaaa=0001 bbbb=0010 → 00110100 00010010 00010011 01000000?
    # 字面解释：0x1234 → 低16位 = 0x1234 → 字节3=0x12 字节2=0x34（大端指令编码）
    assert t.encode("add r1, r2, 0x1234") == bytes.fromhex("34 12 12 34")

def test_label_resolution():
    # label 值 = 字节地址（两条 4 字节指令后 jmp 到 first）
    src = """
first:
    mov r1, 1
    jmp first
    """
    # mov r1, 1 = 00110001 00010000 00000000 00000001 → 31 10 00 01
    # jmp first = 01011000 00001111 00000000 00000100 → 58 0F 00 04
    assert t.encode(src) == bytes.fromhex("31 10 00 01 58 0f 00 04")

def test_data_and_string():
    src = 'U32 0x12345678\n"AB"'
    assert t.encode(src) == bytes.fromhex("12 34 56 78 41 42")

def test_push_pop_call_ret_sizes():
    src = "push r1\npop r2\ncall f\nret\nf:\njmp f"
    b = t.encode(src)
    # push=8B, pop=8B, call=16B, ret=12B, jmp=4B → 总 48 字节
    assert len(b) == 48

- [ ] **Step 3: 实现 `emu/emu.py`**

1. 内存 `bytearray(ram_size)`，程序从地址 0 载入
2. 寄存器数组 16 个 + `pc`；`flags` 寄存器是普通寄存器之一（索引 15），**调用约定约定 flags 由 call/ret 承载返回地址**，模拟器对 flags 无特殊语义
3. 指令循环：按 32 位小端取出指令字 → 依据编码表反查（模式位 bits29-30 + imm 位 bit28 + opcode bits27-24 + 具体指令）确定长度与操作数 → 执行 → `pc += len`
4. 语义实现（对照 .isa 注释逐条）：
   - `nop` 空转；`in`/`keyboard`/`time_0`/`time_1`/`counter` → 目标寄存器置 0（M1 不用）；`out` → 打印寄存器值的调试行（可选）
   - `screen a,b` → 记录 `settings[a] = b`（不渲染）
   - ALU：add/sub/mul/div/mod/nand/nor/and/or/xor/lsl/lsr/asr/cmp（32 位无符号语义；lsl/lsr/asr 位移量取 c 的低 5 位；cmp 置 flags 与条件跳转判据一致——**flags 具体位布局不定义，条件跳转 je/jne/jb/ja/jl/jg 直接依据上一 cmp 的 a/b 值判定**：模拟器内部维护 `last_cmp = (a, b, is_signed)`，条件跳转据此判定，实现简单且与硬件语义一致）
   - 跳转：jmp/条件跳转（imm 型）→ `pc = imm`；jmp(reg) → `pc = reg`
   - load/store 8/16/32（小端读写 mem）；pload/pstore → 内存映射到 `disk` bytearray（模拟器默认空，不影响 M1）
   - push/pop/call/ret：按 Task 2 的展开序列逐条执行即可（无需特殊处理——它们是普通指令）
5. 停机：检测 `jmp 自身地址`（自跳转）→ 停止，`exit_code = regs[1]`；`max_instr` 超限抛 `InfiniteLoopError`
6. `EmuResult`: `.regs`, `.mem`, `.exit_code`, `.settings`

`runtime/divsi3.c`（用 unsigned 除/模 + neg + 比较——全部是 M1 已有特性）：
```c
int __divsi3(int a, int b) {
    unsigned ua = a < 0 ? 0u - (unsigned)a : (unsigned)a;
    unsigned ub = b < 0 ? 0u - (unsigned)b : (unsigned)b;
    unsigned q = ua / ub;
    return (a < 0) != (b < 0) ? 0 - (int)q : (int)q;
}
int __modsi3(int a, int b) {
    unsigned ua = a < 0 ? 0u - (unsigned)a : (unsigned)a;
    unsigned ub = b < 0 ? 0u - (unsigned)b : (unsigned)b;
    unsigned r = ua % ub;
    return a < 0 ? 0 - (int)r : (int)r;
}
```
（注：`a == INT_MIN` 时为 UB，教学可接受，注释说明即可。`0u - (unsigned)a` 用 `neg` 指令——编译器对 `0 - x` 生成 `neg` 或 `sub r1, zr, r2`。）

```c
#include <stdio.h>
#include <string.h>
#include <assert.h>
#include "emu/asm.h"

static void check(const char *src, const char *want, size_t want_len) {
    uint8_t out[4096];
    AsmError err;
    int n = asm_assemble(src, out, sizeof out, &err);
    assert(n == (int)want_len);
    if (want) assert(memcmp(out, want, want_len) == 0);
}

int main(void) {
    /* 1. add r1, r2, r3 → 24 12 03 00（byte0 = 00100100） */
    check("add r1, r2, r3", "\x24\x12\x03\x00", 4);
    /* 2. add r1, r2, 0x1234 → 34 12 12 34（imm 大端占 byte2-3） */
    check("add r1, r2, 0x1234", "\x34\x12\x12\x34", 4);

    check("mov r1, 1\nfirst:\njmp first",
```

- [ ] **Step 3: 实现 `emu/isa.h` + `emu/isa.c`**

编码模型（直接对应 .isa 位模式）：
- 指令字 32 位：**byte0 = bits 31–24**（与 .isa 文本从左到右的 4 组字节一致）；写入字节流时 byte0 在前
- 操作数用 **bit 位置**（字段 LSB 位号）描述：`OP_REG`（4 位）、`OP_IMM`（16 位）；`not` 的 b 字段同时出现在 bit16 与 bit8（双位域，编码时两处同值）

```c
/* emu/isa.h */
#ifndef SYMPLUS_ISA_H
#define SYMPLUS_ISA_H
#include <stdint.h>
#include <stddef.h>

enum { OP_NONE, OP_REG, OP_IMM };
enum { MAX_OPS = 3, MAX_WORDS = 5 };      /* call = 5 词 */
#define MAX_INSN_BYTES (MAX_WORDS * 4)

typedef struct { int kind; int pos[2]; } OpField; /* pos[1]=-1 或第二个位域（not） */

typedef struct {
    const char *name;
    int nwords;                 /* 1 = 普通指令；2/3/5 = 伪指令 */
    uint32_t bits[MAX_WORDS];   /* 每词固定位 */
    uint32_t mask[MAX_WORDS];   /* 每词固定位掩码（字段位为 0） */
    OpField ops[MAX_WORDS][3];  /* 每词操作数位域 */
    int opcount;                /* 总操作数 */
} Insn;

extern const Insn isa_table[];
extern const int isa_table_len;
const Insn *isa_find(const char *name);
int isa_encode(const Insn *insn, const uint32_t *vals, uint8_t out[MAX_INSN_BYTES]); /* 返回总字节数 */
int isa_reg_index(const char *name); /* zr..flags → 0..15，非寄存器返回 -1 */
#endif
```

转写清单（对照 .isa 逐条核对，**76 条**）：nop；in；out(reg)；out(imm)；keyboard；screen(reg)；screen(imm)；time_0；time_1；counter；nand/or/and/nor/add/sub/xor/lsl/lsr/asr/mul/div/mod × (reg + imm)（13+13=26 条，三操作数）；cmp(reg) + cmp(imm)（2 条，byte1 固定 1111 前缀）；jmp(reg)；jmp/je/jne/jb/jae/jbe/ja/jl/jge/jle/jg(imm)（12 条）；load_8/16/32、pload × (reg + imm)（8 条）；store_8/16/32、pstore × (reg + imm)（8 条）；mov(reg)、mov(imm)；neg(reg)、neg(imm)；not(reg)、not(imm)；push、pop、call、ret（4 条伪指令）。共 72 单指令 + 4 伪指令 = **76 条**（旧计划 66 条漏数，已更正）。

**表内顺序注意**：mov/neg/not 排在 or/sub/nor 之前——mov 的编码是 or 编码的严格子集（mov ≡ or %a,%b,zr，等价 `or r1, r2, r2` 之类也是 mov 语义），按表序先命中更特定的别名，行为一致（Task 3 模拟器按 byte0 分发亦如此）。

流程：逐行处理 → 去注释（`;` 与 `//`，**字符串字面量内的 `;` 不截断**）→ 识别 `name:` label（记录字节偏移；可与指令同行）→ 数据行 `U8/U16/U32/U64 <值>`（大端发射）、字符串 `"..."`（原样字节）、`@0x…` 填充（补 0x00 至目标地址）→ 指令行按助记符查表：同名多变体按操作数类型匹配（`add r1,r2,r3` 命中 reg 型，`add r1,r2,5` 命中 imm 型——寄存器名是保留字，解析无歧义）；支持字符字面量 `'A'` → 0x41。两遍：第一遍算字节偏移（指令定长 4/8/12/20B），第二遍把 label 引用解析为字节地址填入 IMM 字段；**立即数或 label > 0xFFFF 硬报错**（对应游戏汇编器 "Immediate too large"，不截断）；未定义 label 报错。出错带行号填入 `AsmError`。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <assert.h>
#include "emu/emu.h"
#include "emu/asm.h"

static EmuResult run_src(const char *src) {
    uint8_t bin[4096];
    AsmError err;
    int n = asm_assemble(src, bin, sizeof bin, &err);
    assert(n > 0);
    EmuResult r = emu_run(bin, (size_t)n, 1 << 20, 10000000);
    assert(r.error == 0);
    return r;
}

int main(void) {
    /* 1. r1 = 2 + 3 */
    EmuResult r = run_src("mov r1, 2\nmov r2, 3\nadd r1, r1, r2\nhalt:\njmp halt");
    assert(r.regs[1] == 5);

    /* 2. 手写调用序列：main 调 sub（全栈传参），sub 返回 40*2=80 */
    r = run_src(
        "mov sp, 0x4000\n"
        "mov r1, 40\n"
        "sub sp, sp, 4\n"
        "store_32 [sp], r1\n"
        "call sub\n"
        "add sp, sp, 4\n"
        "jmp end\n"
        "sub:\n"
        "    add r9, sp, 4\n"      /* 无 offset 寻址：add 合成地址 */
        "    load_32 r2, [r9]\n"
        "    add r1, r2, r2\n"
        "    ret\n"
        "end:\n"
        "halt:\n"
        "    jmp halt\n");
    assert(r.regs[1] == 80);

    /* 3. 屏幕缓冲：向 0x2000 写 "Hi"（模拟器内存断言） */
    r = run_src(
        "mov r1, 0x2000\n"
        "mov r2, 'H'\n"            /* 字符字面量 → ASCII 72 */
        "store_8 [r1], r2\n"
        "mov r2, 'i'\n"
        "store_8 [r1], r2\n"
        "halt:\n"
        "    jmp halt\n");
    assert(memcmp(r.mem + 0x2000, "Hi", 2) == 0);

    /* 4. 停机约定：自跳转停止，exit_code = r1 */
    r = run_src("mov r1, 42\nhalt:\njmp halt");
    assert(r.exit_code == 42);

    printf("ALL PASS\n");
    return 0;
}
```

- [ ] **Step 3: 实现 `emu/emu.c`**

1. 内存 `calloc(ram_size)`，二进制从地址 0 载入；`disk` 数组（1MB，全 0，M1 不用）
2. 寄存器数组 16 个 + `pc`；**指令字按大端解码**（byte0 = bits 31–24），数据读写按**小端**（与游戏内 RAM 配置一致）
3. 指令循环：取 4 字节合成 32 位字 → **按 byte0 分发**（每类语义唯一：0x21=or/mov、0x25=sub/neg、0x23=nor/not、0x31/0x33/0x35 同理，别名编码行为一致；伪指令展开词本身就是普通指令——push 词0=sub sp,sp,4、call 词4=jmp imm 等，见 .isa 等价注释）→ 执行 → `pc += 4`
4. 语义（对照 .isa 注释逐条）：
   - `nop` 空转；`in`/`keyboard`/`time_0`/`time_1` → 目标寄存器置 0（M1 不用）；`out` → 打印数值调试行到 stdout（可选中关闭）
   - `screen a,b` → `settings[a] = b`（不渲染）
   - ALU：add/sub/mul/div/mod/nand/nor/and/or/xor/lsl/lsr/asr（32 位无符号语义；lsl/lsr/asr 位移量取 c 的低 5 位；div/mod 除数为 0 → 结果 0）
   - `counter` → 写入**自身指令地址**（call 展开的 `add flags,flags,20` 依赖此，20 = call 总长，pc+20 恰为 call 之后第一条指令）
   - `cmp` → 记录 `last_cmp = (a, b)`（**flags 寄存器位布局不定义**；条件跳转直接依据 last_cmp 判定，与硬件语义一致且自洽；flags 寄存器仍按普通寄存器参与运算——call/ret 用它承载返回地址，天然支持）
   - 跳转：jmp/条件跳转（imm 型）→ `pc = imm`；jmp(reg) → `pc = reg`；**自跳转（目标 == 当前指令地址）→ 停机**，`exit_code = regs[1]`；`max_instr` 超限 → `error = 1`
   - load/store 8/16/32（小端读写 mem）；pload/pstore → 读写 `disk`；地址越界 → `error = 2`
   - push/pop/call/ret **无需特殊处理**——按展开序列逐条执行即可
5. `EmuResult` 组装返回

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <assert.h>
#include "emu/emu.h"
#include "emu/asm.h"

#define SYMCC "symcc/symcc.exe"

/* 编译（symcc.exe）→ 汇编（进程内 asm_assemble）→ 模拟（emu_run）→ 断言 */
static void compile_and_run(const char *c_file, int expected_exit,
                            const char *screen, size_t screen_len,
                            const char *extra_file) {
    char cmd[512], asm_file[256], text[1 << 20];
    uint8_t bin[1 << 20];
    AsmError aerr;
    EmuResult res;
    int n;
    FILE *fp;
    size_t tlen;

    if (extra_file && extra_file[0])
        snprintf(cmd, sizeof cmd, "%s %s %s -o %s.asm", SYMCC, c_file, extra_file, c_file);
    else
        snprintf(cmd, sizeof cmd, "%s %s -o %s.asm", SYMCC, c_file, c_file);
    assert(system(cmd) == 0);

    snprintf(asm_file, sizeof asm_file, "%s.asm", c_file);
    fp = fopen(asm_file, "rb");
    assert(fp != NULL);
    tlen = fread(text, 1, sizeof text - 1, fp);
    fclose(fp);
    text[tlen] = 0;

    n = asm_assemble(text, bin, sizeof bin, &aerr);
    assert(n > 0);
    res = emu_run(bin, (size_t)n, 1 << 20, 100000000);
    assert(res.error == 0);
    assert(res.exit_code == expected_exit);
    if (screen_len)
        assert(memcmp(res.mem + 0x2000, screen, screen_len) == 0);
    printf("PASS %s (exit=%d)\n", c_file, res.exit_code);
    free(res.mem);
}

int main(void) {
    compile_and_run("tests/test_arith.c", 7, NULL, 0, "");
    printf("ALL TESTS PASSED\n");
    return 0;
}
```

```c
int __divsi3(int a, int b) {
    unsigned ua = a < 0 ? 0u - (unsigned)a : (unsigned)a;
    unsigned ub = b < 0 ? 0u - (unsigned)b : (unsigned)b;
    unsigned q = ua / ub;
    return (a < 0) != (b < 0) ? 0 - (int)q : (int)q;
}
int __modsi3(int a, int b) {
    unsigned ua = a < 0 ? 0u - (unsigned)a : (unsigned)a;
    unsigned ub = b < 0 ? 0u - (unsigned)b : (unsigned)b;
    unsigned r = ua % ub;
    return a < 0 ? 0 - (int)r : (int)r;
}
```
